<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let y = 350;
let vy = 0;
function draw() {
  vy += 0.5;
  y += vy;
  if (y > 350) { y = 350; vy = 0; }
  ctx.fillStyle = '#331';
  ctx.fillRect(0, 0, 400, 400);
  ctx.fillStyle = '#f80';
  ctx.fillRect(190, y, 20, 20);
}
document.addEventListener('keydown', (e) => {
  if (e.key === ' ' && y >= 350) vy = -10;
});
setInterval(draw, 33);
draw();
</script>
</body>
