<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let y = 0;
let vy = 0;
function resize() {
  canvas.width = canvas.clientWidth;
  canvas.height = canvas.clientHeight;
}
function draw() {
  const ground = canvas.height * 0.85;
  const size = Math.min(canvas.width, canvas.height) / 20;
  vy += canvas.height * 0.0012;
  y += vy;
  if (y > ground - size) { y = ground - size; vy = 0; }
  ctx.fillStyle = '#331';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = '#f80';
  ctx.fillRect(canvas.width / 2 - size / 2, y, size, size);
}
document.addEventListener('keydown', (e) => {
  if (e.key === ' ') vy = -canvas.height * 0.025;
});
addEventListener('resize', resize);
resize();
setInterval(draw, 33);
draw();
</script>
</body>
</html>
